import httpx
from app.config import settings

async def fetch_threads_jobs():
    # Sử dụng endpoint search chính thức của Threads API
    url = "https://graph.threads.net/v1.0/threads/search"
    headers = {"Authorization": f"Bearer {settings.THREADS_API_TOKEN}"}
    
    # Định nghĩa các từ khóa ngách chất lượng cao
    keywords = ["looking for dev", "hiring freelancer", "need a landing page", "hire ghostwriter"]
    all_jobs = []

    async with httpx.AsyncClient() as client:
        for query in keywords:
            params = {"q": query, "fields": "id,text,permalink,timestamp,username"}
            response = await client.get(url, headers=headers, params=params)
            
            if response.status_code == 200:
                posts = response.json().get("data", [])
                for post in posts:
                    # Normalize dữ liệu chuẩn theo cấu trúc AI-FOS /ingest
                    job_data = {
                        "source": "threads",
                        "external_url": post["permalink"],
                        "title": f"Threads job from @{post['username']}",
                        "description_raw": post["text"]
                    }
                    all_jobs.append(job_data)
    return all_jobs

import httpx
from app.config import settings

async def fetch_youtube_jobs():
    # Endpoint tìm kiếm hoạt động mới nhất của các kênh mục tiêu
    url = "https://www.googleapis.com/youtube/v3/search"
    
    # Danh sách các channel ID của các nhà sáng lập/tech creator bạn nhắm tới
    target_channels = ["CHANNEL_ID_1", "CHANNEL_ID_2"] 
    all_jobs = []

    async with httpx.AsyncClient() as client:
        for channel_id in target_channels:
            params = {
                "key": settings.YOUTUBE_API_KEY,
                "channelId": channel_id,
                "part": "snippet",
                "order": "date",
                "maxResults": 3,
                "type": "video" # Hoặc quét lấy các hoạt động bài post
            }
            response = await client.get(url, params=params)
            if response.status_code == 200:
                items = response.json().get("items", [])
                for item in items:
                    description = item["snippet"]["description"]
                    title = item["snippet"]["title"]
                    video_id = item["id"]["videoId"]
                    
                    # Kiểm tra xem mô tả video có chứa từ khóa tuyển dụng không
                    if any(word in description.lower() for word in ["hiring", "looking for", "contact me at"]):
                        job_data = {
                            "source": "youtube",
                            "external_url": f"https://www.youtube.com/watch?v={video_id}",
                            "title": f"YouTube Opportunity: {title}",
                            "description_raw": description
                        }
                        all_jobs.append(job_data)
    return all_jobs
